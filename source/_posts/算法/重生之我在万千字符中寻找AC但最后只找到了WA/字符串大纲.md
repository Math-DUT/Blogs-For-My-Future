但是作为一直在吃C++字符串操作的shi的ACMer，字符串题也是那些逆天命题组常考的东西

在做题之前，先看看C++以及Python里面字符串有哪些操作：

```cpp
#include <bits/stdc++.h>
using namespace std;
int main() {
    string s = "abcabc", t = "bc";
    // 1. 长度、判空、访问
    s.size();                 // 长度
    s.empty();                // 是否为空
    s[0];                     // 第一个字符
    s.back();                 // 最后一个字符
    // 2. 遍历
    for (char c : s) cout << c;
    for (int i = 0; i < (int)s.size(); i++)
        cout << s[i];
    // 3. 拼接、增删字符
    s += t;                   // 拼接
    s.push_back('x');         // 末尾添加字符
    s.pop_back();             // 删除末尾字符
    // 4. 截取子串：起点、长度
    string sub = s.substr(1, 3);
    // 5. 查找子串
    size_t pos = s.find(t);
    if (pos != string::npos) cout << pos;
    // 找不到返回 string::npos
    // 6. 替换、删除
    string a = "abcabc";
    a.replace(0, 2, "XY");    // 指定位置替换
    a.erase(0, 2);            // 从下标 0 删除 2 个字符
    string b = "banana";
    b.erase(remove(b.begin(), b.end(), 'a'), b.end());
    // 删除所有 'a'
    // 7. 反转、排序
    string rev = s;
    reverse(rev.begin(), rev.end());
    string sorted = s;
    sort(sorted.begin(), sorted.end());
    // 8. 字符统计
    int cnt[256] = {};
    for (unsigned char c : s) cnt[c]++;
    cout << cnt['a'];
    // 9. 大小写转换
    for (char &c : s)
        c = (char)tolower((unsigned char)c);
    // 转小写；toupper(c) 可转大写
    // 10. 字符与数字转换
    int digit = '7' - '0';    // 7
    char ch = 5 + '0';        // '5'
    int x = stoi("123");      // 字符串转整数
    string num = to_string(x);// 整数转字符串
    // 11. 字符判断（需要 <cctype>）
    isdigit('7');             // 数字字符
    isalpha('a');             // 字母
    isalnum('a');             // 字母或数字
    islower('a');             // 小写字母
    isupper('A');             // 大写字母
    // 12. 分割字符串（按空白分割）
    string line = "apple banana orange";
    stringstream ss(line);
    string word;
    while (ss >> word)
        cout << word << '\n';
    // 13. 判断回文串
    string p = "abba", q = p;
    reverse(q.begin(), q.end());
    bool palindrome = (p == q);
    // 14.插入字符
    s.insert(0, "XY");    // 插入字符串 "XY"
	s.insert(0, 2, 'X'); // 插入两个字符 'X'
    return 0;
}
```
